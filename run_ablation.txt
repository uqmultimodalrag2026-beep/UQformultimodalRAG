sbatch <<'EOF'
#!/bin/bash
#SBATCH --job-name=lars_ablated_infer
#SBATCH --time=1-00:00:00
#SBATCH --gpus=1
#SBATCH --mem=160000
#SBATCH --partition=H100,H100-SEE,H200,RTXA6000
#SBATCH --array=0-11
#SBATCH --output=/log_root_dir/lars_ablated_infer_%A_%a.out
#SBATCH --error=/log_root_dir/lars_ablated_infer_%A_%a.err

set -euo pipefail

# =========================
# USER CONFIG
# =========================
LOG_DIR="/log_root_dir"
RESULTS_ROOT="/result_root_dir/experiment_results_ablated"
DATA_ROOT="/dataset_root_dir"
MODEL_ROOT="/model_root_dir"
ABLATED_MODEL_ROOT="${MODEL_ROOT}/ablated"
SCRATCH_ROOT="/scratch_root_dir"
FSCRATCH_ROOT="/fast_scratch_root_dir"
CONTAINER_IMAGE="/container_image.sqsh"
INSTALL_SCRIPT="install_qwen.sh"

EVQA_DATASET_DIR="${DATA_ROOT}/evqa"
EVQA_IMAGE_DIR="${EVQA_DATASET_DIR}/images/raw"
EVQA_DATASET_CSV="evqa_test.csv"
EVQA_KB_PATH="${EVQA_DATASET_DIR}/"

INFOSEEK_DATASET_DIR="${DATA_ROOT}/infoseek"
INFOSEEK_IMAGE_DIR="${INFOSEEK_DATASET_DIR}/images/"
INFOSEEK_DATASET_CSV="infoseek_test.csv"

LLAVA_MODEL="llava-hf/llava-1.5-7b-hf"
QWEN_MODEL="Qwen/Qwen3-VL-4B-Instruct"

SAMPLES=2000
SAMPLES_PER_RUN=2000

mkdir -p "${LOG_DIR}" "${RESULTS_ROOT}"

CONFIGS=(
  "evqa llava bm25"
  "evqa llava rerank"
  "evqa llava eva_clip+contriever"
  "evqa qwen bm25"
  "evqa qwen rerank"
  "evqa qwen eva_clip+contriever"
  "infoseek llava bm25"
  "infoseek llava rerank"
  "infoseek llava eva_clip+contriever"
  "infoseek qwen bm25"
  "infoseek qwen rerank"
  "infoseek qwen eva_clip+contriever"
)

read DATASET MODEL_FAMILY CONTEXT_MODE <<< "${CONFIGS[$SLURM_ARRAY_TASK_ID]}"

case "$MODEL_FAMILY" in
  qwen) CHAT_MODEL="${QWEN_MODEL}" ;;
  llava) CHAT_MODEL="${LLAVA_MODEL}" ;;
  *) echo "Unknown model family: $MODEL_FAMILY"; exit 1 ;;
esac

case "$DATASET" in
  evqa)
    DATASET_PATH="${EVQA_DATASET_DIR}"
    IMAGE_DIR="${EVQA_IMAGE_DIR}"
    DATASET_CSV="${EVQA_DATASET_CSV}"
    MODEL_DATASET_DIR="${DATASET}_${MODEL_FAMILY}"
    case "$CONTEXT_MODE" in
      bm25) RETRIEVAL_CSV="bm25_test.csv"; RETRIEVAL_TAG="bm25" ;;
      rerank) RETRIEVAL_CSV="rerank_test.csv"; RETRIEVAL_TAG="rerank" ;;
      eva_clip+contriever) RETRIEVAL_CSV="eva_clip_test.csv"; RETRIEVAL_TAG="eva_clip" ;;
      *) echo "Unknown evqa context mode: $CONTEXT_MODE"; exit 1 ;;
    esac
    ;;
  infoseek)
    DATASET_PATH="${INFOSEEK_DATASET_DIR}"
    IMAGE_DIR="${INFOSEEK_IMAGE_DIR}"
    DATASET_CSV="${INFOSEEK_DATASET_CSV}"
    MODEL_DATASET_DIR="infoseek_new_${MODEL_FAMILY}"
    case "$CONTEXT_MODE" in
      bm25) RETRIEVAL_CSV="bm25_test.csv"; RETRIEVAL_TAG="bm25" ;;
      rerank) RETRIEVAL_CSV="rerank_test.csv"; RETRIEVAL_TAG="rerank" ;;
      eva_clip+contriever) RETRIEVAL_CSV="eva_clip_test.csv"; RETRIEVAL_TAG="eva_clip" ;;
      *) echo "Unknown infoseek context mode: $CONTEXT_MODE"; exit 1 ;;
    esac
    ;;
  *) echo "Unknown dataset: $DATASET"; exit 1 ;;
esac

BASE_MODEL_PATH_NO_CONTEXT="${MODEL_ROOT}/${MODEL_DATASET_DIR}/lars_${RETRIEVAL_TAG}"
BASE_MODEL_PATH_WITH_CONTEXT="${MODEL_ROOT}/${MODEL_DATASET_DIR}/lars_modified_${RETRIEVAL_TAG}_with_context"

JOB_LABEL="${DATASET}_${MODEL_FAMILY}_${CONTEXT_MODE}_ablated"
scontrol update JobId="${SLURM_JOB_ID}" JobName="${JOB_LABEL}" >/dev/null 2>&1 || true

OUTDIR="${RESULTS_ROOT}/${DATASET}_${MODEL_FAMILY}_${CONTEXT_MODE}"
mkdir -p "${OUTDIR}"

ABL_NO_P_PRIME="${ABLATED_MODEL_ROOT}/${DATASET}_lars_ablated_${MODEL_FAMILY}_${RETRIEVAL_TAG}_no_p_prime"
ABL_NO_P_Q_C="${ABLATED_MODEL_ROOT}/${DATASET}_lars_ablated_${MODEL_FAMILY}_${RETRIEVAL_TAG}_no_p_q_c"
ABL_NO_P_Q_I="${ABLATED_MODEL_ROOT}/${DATASET}_lars_ablated_${MODEL_FAMILY}_${RETRIEVAL_TAG}_no_p_q_i"
ABL_NO_P_Q="${ABLATED_MODEL_ROOT}/${DATASET}_lars_ablated_${MODEL_FAMILY}_${RETRIEVAL_TAG}_no_p_q"

LARS_DICTS=$(cat <<JSON
[
  {"lars_name": "${MODEL_FAMILY}_${DATASET}_${RETRIEVAL_TAG}_base_no_context", "lars_version": "lars_finetuned", "model_path": "${BASE_MODEL_PATH_NO_CONTEXT}", "lars_with_context": false},
  {"lars_name": "${MODEL_FAMILY}_${DATASET}_${RETRIEVAL_TAG}_base_with_context", "lars_version": "lemuq", "model_path": "${BASE_MODEL_PATH_WITH_CONTEXT}", "lars_with_context": true},
  {"lars_name": "${MODEL_FAMILY}_${DATASET}_${RETRIEVAL_TAG}_no_p_prime", "lars_version": "lemuq_ablated", "model_path": "${ABL_NO_P_PRIME}", "lars_with_context": true, "p_prime": false, "p_q_i": true, "p_q_c": true, "p_q": true},
  {"lars_name": "${MODEL_FAMILY}_${DATASET}_${RETRIEVAL_TAG}_no_p_q_i", "lars_version": "lemuq_ablated", "model_path": "${ABL_NO_P_Q_I}", "lars_with_context": true, "p_prime": true, "p_q_i": false, "p_q_c": true, "p_q": true},
  {"lars_name": "${MODEL_FAMILY}_${DATASET}_${RETRIEVAL_TAG}_no_p_q_c", "lars_version": "lemuq_ablated", "model_path": "${ABL_NO_P_Q_C}", "lars_with_context": true, "p_prime": true, "p_q_i": true, "p_q_c": false, "p_q": true},
  {"lars_name": "${MODEL_FAMILY}_${DATASET}_${RETRIEVAL_TAG}_no_p_q", "lars_version": "lemuq_ablated", "model_path": "${ABL_NO_P_Q}", "lars_with_context": true, "p_prime": true, "p_q_i": true, "p_q_c": true, "p_q": false}
]
JSON
)

echo "Dataset=${DATASET} Model=${MODEL_FAMILY} Context=${CONTEXT_MODE} Output=${OUTDIR}"

srun \
  --container-image="${CONTAINER_IMAGE}" \
  --container-workdir="$(pwd)" \
  --container-mounts="${SCRATCH_ROOT}:${SCRATCH_ROOT},$(pwd):$(pwd),${FSCRATCH_ROOT}:${FSCRATCH_ROOT}" \
  bash -c "
    set -e
    bash ${INSTALL_SCRIPT}
    cd src
    python3 main.py \
      --truth_methods lars \
      --model ${CHAT_MODEL} \
      --dataset ${DATASET} \
      --context_mode ${CONTEXT_MODE} \
      --samples ${SAMPLES} \
      --samples_per_run ${SAMPLES_PER_RUN} \
      --output_dir ${OUTDIR} \
      --no_8bit \
      --dataset_path ${DATASET_PATH} \
      --kb_path ${EVQA_KB_PATH} \
      --dataset_csv ${DATASET_CSV} \
      --retrieval_csv ${RETRIEVAL_CSV} \
      --image_directory ${IMAGE_DIR} \
      --lars_dicts '${LARS_DICTS}'
  "
EOF
